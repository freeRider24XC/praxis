# ZhiXing主题定制示例

## 🎨 基础主题定制

### 1. 修改品牌色彩
```dart
// lib/common/style/custom_themes.dart
import 'package:flutter/material.dart';

class ZhiXingTheme {
  // 自定义品牌色彩
  static const Color primaryColor = Color(0xFF1E88E5);      // 深蓝色
  static const Color secondaryColor = Color(0xFF26A69A);    // 青绿色
  static const Color accentColor = Color(0xFFFF6F00);       // 橙色
  static const Color successColor = Color(0xFF43A047);      // 绿色
  static const Color warningColor = Color(0xFFFFB300);      // 黄色
  static const Color errorColor = Color(0xFFE53935);       // 红色

  static LinearGradient get primaryGradient => LinearGradient(
    colors: [primaryColor, Color(0xFF1565C0)],
    begin: Alignment.topLeft,
    end: Alignment.bottomRight,
  );
  
  static LinearGradient get successGradient => LinearGradient(
    colors: [successColor, Color(0xFF2E7D32)],
    begin: Alignment.topLeft,
    end: Alignment.bottomRight,
  );
}
```

## 📱 响应式主题

### 不同屏幕尺寸的主题适配
```dart
class ResponsiveTheme {
  static ThemeData getResponsiveTheme(BuildContext context) {
    final screenWidth = MediaQuery.of(context).size.width;
    final theme = Theme.of(context);
    
    if (screenWidth > 1200) {
      // 桌面端 - 使用更大的圆角和更强的阴影
      return theme.copyWith(
        cardTheme: CardTheme(
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(20),
          ),
          elevation: 4,
        ),
      );
    } else if (screenWidth > 600) {
      // 平板端 - 中等样式
      return theme.copyWith(
        cardTheme: CardTheme(
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(16),
          ),
          elevation: 3,
        ),
      );
    } else {
      // 手机端 - 保持默认样式
      return theme;
    }
  }
}
```

## 🎨 动态主题

### 根据时间切换主题
```dart
class TimeBasedTheme {
  static ThemeData getTimeBasedTheme() {
    final hour = DateTime.now().hour;
    
    if (hour >= 6 && hour < 18) {
      // 白天 - 使用浅色主题
      return ZhiXingTheme.lightTheme;
    } else {
      // 夜晚 - 使用深色主题
      return ZhiXingDarkTheme.darkTheme;
    }
  }
}

// 使用示例
class ThemeController extends GetxController {
  final themeMode = ThemeMode.system.obs;
  
  void toggleTimeBasedTheme() {
    final theme = TimeBasedTheme.getTimeBasedTheme();
    Get.changeTheme(theme);
  }
}
```

## 🔧 主题切换动画

### 主题切换动画效果
```dart
class AnimatedThemeSwitcher extends StatefulWidget {
  final Widget child;
  
  const AnimatedThemeSwitcher({Key? key, required this.child}) : super(key: key);
  
  @override
  _AnimatedThemeSwitcherState createState() => _AnimatedThemeSwitcherState();
}

class _AnimatedThemeSwitcherState extends State<AnimatedThemeSwitcher>
    with SingleTickerProviderStateMixin {
  late AnimationController _animationController;
  late Animation<double> _fadeAnimation;
  
  @override
  void initState() {
    super.initState();
    _animationController = AnimationController(
      duration: Duration(milliseconds: 300),
      vsync: this,
    );
    _fadeAnimation = Tween<double>(
      begin: 0.0,
      end: 1.0,
    ).animate(CurvedAnimation(
      parent: _animationController,
      curve: Curves.easeInOut,
    ));
  }
  
  @override
  Widget build(BuildContext context) {
    return FadeTransition(
      opacity: _fadeAnimation,
      child: widget.child,
    );
  }
  
  @override
  void dispose() {
    _animationController.dispose();
    super.dispose();
  }
}

// 主题切换服务
class ThemeSwitcherService {
  static void switchTheme(BuildContext context) {
    final currentTheme = Theme.of(context).brightness;
    final newThemeMode = currentTheme == Brightness.light 
        ? ThemeMode.dark 
        : ThemeMode.light;
    
    Get.changeThemeMode(newThemeMode);
  }
}
```

## 🎭 个性化主题

### 用户可定制的主题
```dart
class UserCustomTheme {
  // 用户偏好设置
  static const String userThemeKey = 'user_custom_theme';
  
  // 保存用户主题偏好
  static Future<void> saveUserTheme(UserTheme userTheme) async {
    final prefs = await SharedPreferences.getInstance();
    final themeJson = jsonEncode(userTheme.toJson());
    await prefs.setString(userThemeKey, themeJson);
  }
  
  // 获取用户主题偏好
  static Future<UserTheme?> getUserTheme() async {
    final prefs = await SharedPreferences.getInstance();
    final themeJson = prefs.getString(userThemeKey);
    if (themeJson != null) {
      return UserTheme.fromJson(jsonDecode(themeJson));
    }
    return null;
  }
  
  // 生成用户自定义主题
  static ThemeData generateCustomTheme(UserTheme userTheme) {
    return ThemeData(
      useMaterial3: true,
      colorScheme: ColorScheme.fromSeed(
        seedColor: userTheme.primaryColor,
        brightness: userTheme.isDark ? Brightness.dark : Brightness.light,
      ),
      // 应用用户自定义的样式
      cardTheme: CardTheme(
        shape: RoundedRectangleBorder(
          borderRadius: BorderRadius.circular(userTheme.borderRadius),
        ),
        elevation: userTheme.elevation.toDouble(),
      ),
    );
  }
}

// 用户主题数据模型
class UserTheme {
  final Color primaryColor;
  final Color secondaryColor;
  final bool isDark;
  final double borderRadius;
  final int elevation;
  final String fontFamily;
  
  UserTheme({
    required this.primaryColor,
    required this.secondaryColor,
    required this.isDark,
    required this.borderRadius,
    required this.elevation,
    required this.fontFamily,
  });
  
  Map<String, dynamic> toJson() => {
    'primaryColor': primaryColor.value,
    'secondaryColor': secondaryColor.value,
    'isDark': isDark,
    'borderRadius': borderRadius,
    'elevation': elevation,
    'fontFamily': fontFamily,
  };
  
  factory UserTheme.fromJson(Map<String, dynamic> json) => UserTheme(
    primaryColor: Color(json['primaryColor']),
    secondaryColor: Color(json['secondaryColor']),
    isDark: json['isDark'],
    borderRadius: json['borderRadius'],
    elevation: json['elevation'],
    fontFamily: json['fontFamily'],
  );
}
```

```dart
class ThemeController extends GetxController {
  final themeMode = ThemeMode.system.obs;
  
  void toggleTheme() {
    if (themeMode.value == ThemeMode.light) {
      themeMode.value = ThemeMode.dark;
      Get.changeThemeMode(ThemeMode.dark);
    } else {
      themeMode.value = ThemeMode.light;
      Get.changeThemeMode(ThemeMode.light);
    }
  }
}
```
