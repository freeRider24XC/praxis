## 🎨 在页面中的使用

### 在主页中使用
```dart
class HomePage extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('ZhiXing'),
        actions: [
          IconButton(
            icon: Icon(Icons.brightness_6),
            onPressed: () => _toggleTheme(),
          ),
        ],
      ),
      body: SingleChildScrollView(
        padding: EdgeInsets.all(16),
        child: Column(
          children: [
            // 统计卡片行
            Row(
              children: [
                Expanded(
                  child: ZhiXingStatCard(
                    title: '今日完成',
                    value: '5',
                    subtitle: '任务',
                    icon: Icons.check_circle,
                  ),
                ),
                SizedBox(width: 16),
                Expanded(
                  child: ZhiXingStatCard(
                    title: '进行中',
                    value: '3',
                    subtitle: '任务',
                    icon: Icons.pending,
                  ),
                ),
              ],
            ),
            SizedBox(height: 24),
            
            // 快捷操作卡片
            ZhiXingIconCard(
              title: '添加新任务',
              subtitle: '快速创建任务',
              icon: Icons.add_circle_outline,
              onTap: () => _showAddTaskDialog(),
            ),
            
            ZhiXingIconCard(
              title: 'AI智能规划',
              subtitle: '让AI帮您规划',
              icon: Icons.auto_awesome,
              onTap: () => _openAIChat(),
            ),
            SizedBox(height: 24),
            
            // 今日任务列表
            Text(
              '今日任务',
              style: Theme.of(context).textTheme.headlineSmall,
            ),
            SizedBox(height: 16),
            
            // 任务卡片列表
            ..._getTodayTasks().map((task) => Padding(
              padding: EdgeInsets.only(bottom: 12),
              child: ZhiXingTaskCard(
                title: task.title,
                description: task.description,
                isCompleted: task.isCompleted,
                priority: task.priority,
                dueDate: task.dueDate,
                tags: task.tags,
                onToggleComplete: () => _toggleTask(task.id),
              ),
            )).toList(),
          ],
        ),
      ),
    );
  }
  
  void _toggleTheme() {
    final theme = Get.theme.brightness;
    if (theme == Brightness.light) {
      Get.changeThemeMode(ThemeMode.dark);
    } else {
      Get.changeThemeMode(ThemeMode.light);
    }
  }
  
  void _showAddTaskDialog() {
    Get.dialog(
      Dialog(
        child: Padding(
          padding: EdgeInsets.all(16),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text('添加新任务'),
              SizedBox(height: 16),
              // 添加任务表单
              ElevatedButton(
                onPressed: () {
                  Get.back();
                },
                child: Text('保存'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
```

## 📱 响应式设计

### 在不同屏幕尺寸下的使用
```dart
class ResponsiveTaskCard extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    final screenWidth = MediaQuery.of(context).size.width;
    
    if (screenWidth > 600) {
      // 平板或桌面端 - 横向布局
      return ZhiXingCard(
        child: Row(
          children: [
            Expanded(child: ZhiXingTaskCard(...)),
            SizedBox(width: 16),
            Expanded(child: ZhiXingTaskCard(...)),
          ],
        ),
      );
    } else {
      // 手机端 - 纵向布局
      return Column(
        children: [
          ZhiXingTaskCard(...),
          SizedBox(height: 12),
          ZhiXingTaskCard(...),
        ],
      );
    }
  }
}
```

### 修改卡片主题
```dart
class CustomZhiXingCard extends ZhiXingCard {
  const CustomZhiXingCard({super.key, required super.child, super.onTap});
  
  @override
  Widget build(BuildContext context) {
    return Card(
      color: Theme.of(context).colorScheme.primaryContainer,
      elevation: 4,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(16),
      ),
      child: InkWell(
        onTap: onTap,
        borderRadius: BorderRadius.circular(16),
        child: Padding(
          padding: EdgeInsets.all(20),
          child: child,
        ),
      ),
    );
  }
}
```

### 2. 状态管理
```dart
class TaskCardController extends GetxController {
  final tasks = <Task>[].obs;
  
  void toggleTask(String taskId) {
    final task = tasks.firstWhere((t) => t.id == taskId);
    task.isCompleted = !task.isCompleted;
    tasks.refresh();
  }
}
```

### 3. 错误处理
```dart
ZhiXingCard(

  child: Text('任务内容'),
```
